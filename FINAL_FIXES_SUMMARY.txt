NEOFORGE 1.21.1 CORRECT PATTERN:
================================================================================

The proper NeoForge 1.21.1 block entity GUI pattern is:

// 1. Create MenuType class
public class VendingMachineMenuType extends MenuType<VendingMachineScreenHandler> {
    public VendingMachineMenuType(int id) { super(id); }
    
    @Override
    public AbstractContainerMenu create(Inventory inv, Player player) {
        return new VendingMachineScreenHandler(this.id(), inv, null);
    }
}

// 2. BlockEntity implements MenuProvider
public class VendingMachineBlockEntity extends BlockEntity implements MenuProvider {
    @Override
    public Component getDisplayName() { ... }
    
    @Override
    public AbstractContainerMenu createMenu(int id, Inventory inv, Player player) {
        return new VendingMachineScreenHandler(id, inv, this.vendingInventory);
    }
}

// 3. Block overrides useWithoutItem()
public class VendingMachineBlock extends BaseEntityBlock {
    @Override
    protected InteractionResult useWithoutItem(BlockState state, Level level, 
                                               BlockPos pos, Player player, BlockHitResult hit) {
        if (!level.isClientSide() && player instanceof ServerPlayer sp) {
            if (level.getBlockEntity(pos) instanceof VendingMachineBlockEntity vmbe) {
                sp.openMenu(vmbe, pos);
            }
        }
        return InteractionResult.sidedSuccess(level.isClientSide());
    }
}

// 4. Screen extends AbstractContainerScreen
public class VendingMachineScreen extends AbstractContainerScreen<VendingMachineScreenHandler> {
    public VendingMachineScreen(VendingMachineScreenHandler menu, Inventory inv, Component title) {
        super(menu, inv, title);
    }
    
    @Override
    protected void renderBg(...) { ... }
}
