NEOFORGE 1.21.1 CORRECT PATTERN:
================================================================================

The proper NeoForge 1.21.1 block entity GUI pattern is:

// 1. BlockEntity implements MenuProvider
public class VendingMachineBlockEntity extends BlockEntity implements MenuProvider {
    @Override
    public Component getDisplayName() { ... }
    
    @Nullable
    @Override
    public AbstractContainerMenu createMenu(int id, Inventory inv, Player player) {
        return new VendingMachineScreenHandler(id, inv, vendingInventory);
    }
}

// 2. Block overrides onUse() for GUI opening
public class VendingMachineBlock extends BaseEntityBlock {
    @Override
    public void onUse(BlockState state, Level level, BlockPos pos, Player player) {
        if (!level.isClientSide()) {
            Minecraft minecraft = Minecraft.getInstance();
            AbstractContainerMenu menu = createMenu(pos, player);
            minecraft.setScreen(new AbstractContainerScreen<>(menu, player.getInventory(), title));
        }
    }
}

// 3. Screen extends AbstractContainerScreen
public class VendingMachineScreen<T extends AbstractContainerMenu> 
    extends AbstractContainerScreen<T> {
    public VendingMachineScreen(T menu, Inventory inv, Component title) {
        super(menu, inv, title);
    }
    
    @Override
    protected void renderBg(...) { ... }
}
